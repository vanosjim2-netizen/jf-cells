    Formules (beginnen met =, scheid met ; of ,)

    SOM GEMIDDELDE MIN MAX MEDIAAN=SOM(A1:A10)
    AANTAL AANTALARG AANTAL.ALS SOM.ALS=AANTAL.ALS(B2:B20;">5,5")
    ALS EN OF NIET ALS.FOUT=ALS(B2>=5.5;"Voldoende";"Onvoldoende")
    AFRONDEN INTEGER REST ABS WORTEL MACHT PI ASELECT=AFRONDEN(C2;1)
    VERT.ZOEKEN=VERT.ZOEKEN("Kaas";A2:C20;3;ONWAAR)
    LENGTE LINKS RECHTS HOOFDLETTERS KLEINE.LETTERS SAMENVOEGEN SPATIES.WISSEN=A1&" "&B1
    VANDAAG NU=VANDAAG()
    Ander blad=Blad2!A1 of ='Mijn blad'!A1
    Vaste celverwijzing=$A$1*B2
